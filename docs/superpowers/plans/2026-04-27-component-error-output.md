> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a universal `error` output port on opt-in components plus a new `ErrorHandler` component (retry-with-backoff + bell alerting) that hooks into a small runtime extension.

**Architecture:** New `ErrorPayload` typed channel + ClassVar opt-in (`error_output_enabled`) + central output auto-injection. Runtime catches exceptions in `Graph._execute_tasks`, runs a retry loop in a new `lfx/graph/retry.py` helper, fires the connected `ErrorHandler` only after exhaustion. New `RunStatus.PARTIAL_SUCCESS` distinguishes "completed with handled errors" from clean success and from `FAILED`.

**Tech Stack:** Python (lfx package + langflow base), SQLModel/Alembic (Postgres), pytest (unit + integration), React/TypeScript/Jest (frontend), existing `UsageAlertNotifier` + `AdminNotification` infrastructure.

```python
from __future__ import annotations

import traceback
from dataclasses import dataclass, field
from datetime import datetime, timezone
from uuid import UUID

_STACK_TRACE_MAX_BYTES = 4096
_TRUNCATION_MARKER = "...[truncated]"


@dataclass(frozen=True)
class ErrorPayload:
    """Structured error emitted by a failing component's `error` output port."""

    error_message: str
    error_type: str
    stack_trace: str
    component_id: str
    component_display_name: str
    flow_id: UUID
    flow_run_id: UUID
    attempt_number: int
    occurred_at: datetime = field(
        default_factory=lambda: datetime.now(timezone.utc),
    )


def build_error_payload(
    exc: BaseException,
    *,
    component_id: str,
    component_display_name: str,
    flow_id: UUID,
    flow_run_id: UUID,
    attempt_number: int,
) -> ErrorPayload:
    """Construct an ErrorPayload from an exception, truncating the trace."""
    raw_trace = "".join(
        traceback.format_exception(type(exc), exc, exc.__traceback__)
    )
    if len(raw_trace) > _STACK_TRACE_MAX_BYTES:
        keep = _STACK_TRACE_MAX_BYTES - len(_TRUNCATION_MARKER)
        trace = raw_trace[:keep] + _TRUNCATION_MARKER
    else:
        trace = raw_trace
    return ErrorPayload(
        error_message=str(exc),
        error_type=type(exc).__name__,
        stack_trace=trace,
        component_id=component_id,
        component_display_name=component_display_name,
        flow_id=flow_id,
        flow_run_id=flow_run_id,
        attempt_number=attempt_number,
    )
```

```python
from __future__ import annotations

import asyncio
import random
from collections.abc import Awaitable, Callable
from dataclasses import dataclass
from enum import Enum
from typing import TypeVar

T = TypeVar("T")


class BackoffStrategy(str, Enum):
    NONE = "none"
    FIXED = "fixed"
    EXPONENTIAL = "exponential"
    EXPONENTIAL_WITH_JITTER = "exponential_with_jitter"


@dataclass(frozen=True)
class RetryConfig:
    max_attempts: int = 3  # 0 = no retry; 1 = one retry after failure; etc.
    strategy: BackoffStrategy = BackoffStrategy.EXPONENTIAL_WITH_JITTER
    base_delay_seconds: float = 1.0
    max_delay_seconds: float = 60.0


def compute_delay_seconds(config: RetryConfig, attempt: int) -> float:
    """Delay before the *next* attempt. `attempt` is the 1-indexed prior attempt."""
    if config.strategy is BackoffStrategy.NONE:
        return 0.0
    if config.strategy is BackoffStrategy.FIXED:
        return min(config.base_delay_seconds, config.max_delay_seconds)
    base = config.base_delay_seconds * (2 ** (attempt - 1))
    capped = min(base, config.max_delay_seconds)
    if config.strategy is BackoffStrategy.EXPONENTIAL_WITH_JITTER:
        return capped * random.uniform(0.5, 1.5)
    return capped


async def run_with_retries(
    op: Callable[[int], Awaitable[T]],
    config: RetryConfig,
) -> T:
    """Execute `op(attempt)` up to `1 + config.max_attempts` times.

    `op` receives the 1-indexed attempt number. If `op` raises, sleep
    `compute_delay_seconds(config, attempt)` then try again, up to
    `config.max_attempts` retries. The final exception propagates.

    `max_attempts=0` means: only the original call, no retries.
    """
    total_attempts = 1 + max(0, config.max_attempts)
    last_exc: BaseException | None = None
    for attempt in range(1, total_attempts + 1):
        try:
            return await op(attempt)
        except BaseException as exc:  # noqa: BLE001
            last_exc = exc
            if attempt == total_attempts:
                raise
            delay = compute_delay_seconds(config, attempt)
            if delay > 0:
                await asyncio.sleep(delay)
    # Unreachable: the loop either returns or raises.
    raise last_exc  # type: ignore[misc]
```

class ErrorHandler(Component):
    """Catches errors emitted by upstream components' error ports.

    Configures retry attempts + backoff, dispatches a single alert per
    handled error, and emits `gave_up` with the original ErrorPayload after
    retries are exhausted.

    Does NOT itself expose an `error` output — exceptions raised inside

import logging
from uuid import UUID

from langflow.services.database.models.admin_notification import NotificationAudience
from langflow.services.notifier.protocol import UsageAlertEvent, UsageAlertNotifier

logger = logging.getLogger(__name__)


def _resolve_audience(
    bell_audience: str,
    flow_owner_id: UUID,
    specific_user_id: UUID | None,
) -> tuple[NotificationAudience, UUID | None]:
    if bell_audience == "Flow owner":
        return NotificationAudience.PLATFORM_ADMIN, flow_owner_id
    if bell_audience == "Org admins":
        return NotificationAudience.PLATFORM_ADMIN, None
    if bell_audience == "Specific user":
        return NotificationAudience.PLATFORM_ADMIN, specific_user_id
    msg = f"Unknown bell_audience: {bell_audience!r}"
    raise ValueError(msg)


def _render_template(template: str, payload: ErrorPayload, *, flow_name: str, org_name: str) -> str:
    return template.format(
        flow_name=flow_name,
        org_name=org_name,
        component_name=payload.component_display_name,
        error_type=payload.error_type,
        error_message=payload.error_message,
        attempt_number=payload.attempt_number,

async def _try_handle_via_error_edge(
    self,
    vertex_id: str,
    exc: Exception,
) -> bool:
    """If the failing vertex has a connected `error` output, run retries and
    invoke ErrorHandler. Return True if handled, False to fall through to
    today's `raise result`.
    """
    from lfx.components.reliability.error_handler import ErrorHandler
    from lfx.graph.retry import BackoffStrategy, RetryConfig, run_with_retries
    from lfx.schema.error_payload import build_error_payload

    failing_vertex = self.get_vertex(vertex_id)
    error_edge = self._find_error_edge(failing_vertex)
    if error_edge is None:
        return False

    handler_vertex = self.get_vertex(error_edge.target_id)
    handler_component = handler_vertex.custom_component
    if not isinstance(handler_component, ErrorHandler):
        return False

    # Read static config from the handler vertex (set at flow build time).
    cfg = self._build_retry_config(handler_component)

    attempt_counter = {"n": 1}
    last_exc: Exception | None = exc

    async def attempt_op(attempt: int):
        nonlocal last_exc
        attempt_counter["n"] = attempt
        if attempt == 1:
            # First attempt already failed (we're here because of `exc`).
            raise last_exc  # type: ignore[misc]
        try:
            await self._emit_sse_event("vertex.retrying", {
                "vertex_id": vertex_id,
                "attempt": attempt,
                "max_attempts": cfg.max_attempts,
            })
            return await failing_vertex._build()
        except Exception as e:  # noqa: BLE001
            last_exc = e
            raise

    try:
        await run_with_retries(attempt_op, cfg)
        await self._emit_sse_event("vertex.retry_succeeded", {
            "vertex_id": vertex_id,
            "succeeded_on_attempt": attempt_counter["n"],
        })
        # Success: the failing vertex's normal outputs fire on the next run-loop tick.
        # No need to invoke the handler — recovery is transparent.
        return True
    except Exception:  # noqa: BLE001
        # Retries exhausted. Build the payload, suppress normal successors,
        # invoke the handler.
        await self._emit_sse_event("vertex.retry_exhausted", {
            "vertex_id": vertex_id,
            "attempts": attempt_counter["n"],
        })
        payload = build_error_payload(
            last_exc or exc,
            component_id=vertex_id,
            component_display_name=failing_vertex.display_name,
            flow_id=self.flow_id,
            flow_run_id=self._run_id,
            attempt_number=attempt_counter["n"],
        )
        await self._suppress_normal_successors(vertex_id)
        await self._invoke_error_handler(handler_vertex, payload)
        await self._record_handled_error(payload)  # Task 10
        return True


def _find_error_edge(self, vertex):
    """Return the edge whose source_handle is the vertex's `error` output."""
    for edge in self.edges:
        if edge.source_id == vertex.id and edge.source_handle.get("name") == "error":
            return edge
    return None
